"""Actual production-via scenes followed by fresh, independent joint-C solves.

The six bounded two-net cases cover both via planes, straight/turn access,
and coincident/separated stacks. They do not qualify arbitrary block graphs,
native/new-layer ownership, manufacturing DRC, or parent timing.
"""
import concurrent.futures
import json
from pathlib import Path
import re
import sys

from capture_field_verify import assess, fv_source
from joint_node_binding import POSITIONS, receive_positions
from joint_rc import assemble, emit_spef, receive
from joint_rc_native import anchor_files, READBACK, VARIANTS
from joint_routed_geometry import compare, parse


CASES = ('via10-turn', 'via10-straight', 'via11-turn',
         'via11-straight', 'stack', 'offset-stack')
CENTERS = {10: 11.09, 11: 13.89, 12: 16.69}


def fixture(case, technology):
    assert case in CASES
    g = technology['geometry_um']; e = technology['electrical']
    assert g['width'] == .8 and g['cut_width'] == .6 and g['enclosure'] == .1
    assert g['metal11_bottom'] == 12.89 and g['metal12_bottom'] == 15.69
    assert g['thickness'] == 2 and g['dielectric_gap'] == .8
    boxes = []; target = []; links = []; paths = []; vias = []
    def wire(layer, a, b, start=0, end=0, net='target'):
        x0, y0 = a; x1, y1 = b
        assert (x0 == x1) != (y0 == y1) and x0 <= x1 and y0 <= y1
        if x0 == x1: xy = [x0-.4, y0-start, x1+.4, y1+end]
        else: xy = [x0-start, y0-.4, x1+end, y1+.4]
        boxes.append((net, [xy[0], xy[1], CENTERS[layer]-1, xy[2], xy[3], CENTERS[layer]+1]))
        point = lambda p, cap: '( %d %d %d )' % (round((p[0]+32)*2000), round((p[1]+32)*2000), round(cap*2000))
        path = 'metal%d %s %s' % (layer, point(a, start), point(b, end))
        if net == 'target': paths.append(path)
        return path
    def via(layer, x, y):
        assert layer in (10, 11)
        vias.append(dict(layer=layer, x=x, y=y))
        for metal in (layer, layer+1):
            z = CENTERS[metal]
            boxes.append(('target', [x-.4, y-.4, z-1, x+.4, y+.4, z+1]))
        boxes.append(('target', [x-.3, y-.3, CENTERS[layer]+1, x+.3, y+.3, CENTERS[layer+1]-1]))
        paths.append('metal%d ( %d %d 0 ) BV1_via%d' % (layer, round((x+32)*2000), round((y+32)*2000), layer))
    def node(name, layer, x, y): target.append((name, [x, y, CENTERS[layer]]))
    def edge(a, b): links.append((a, b))
    if case.startswith('via'):
        lower = int(case[3:5]); straight = case.endswith('straight')
        wire(lower, (-3.2, 0), (3.2, 0)); via(lower, 1.6, 0)
        if straight: wire(lower+1, (1.6, 0), (4.8, 0), start=.4)
        else: wire(lower+1, (1.6, 0), (1.6, 3.2), start=.4)
        for name, layer, x, y in [('left', lower, -2.4, 0), ('lower', lower, 1.6, 0),
                                  ('right', lower, 2.4, 0), ('upper', lower+1, 1.6, 0),
                                  ('end', lower+1, 4 if straight else 1.6, 0 if straight else 2.4)]:
            node(name, layer, x, y)
        links = [('left', 'lower'), ('lower', 'right'), ('lower', 'upper'), ('upper', 'end')]
        near_layer = 12 if lower == 10 else 10; near_y = 0
    else:
        wire(10, (-3.2, 0), (3.2, 0))
        x = 1.6 if case == 'stack' else 0
        via(10, x, 0)
        upper_x = x if case == 'stack' else 3.2
        if case == 'offset-stack': wire(11, (x, 0), (upper_x, 0), start=.4, end=.4)
        via(11, upper_x, 0); wire(12, (upper_x, 0), (upper_x, 3.2), start=.4)
        for name, layer, px, py in [('left', 10, -2.4, 0), ('lower', 10, x, 0),
                                    ('right', 10, 2.4, 0), ('middle', 11, x, 0)]:
            node(name, layer, px, py)
        links = [('left', 'lower'), ('lower', 'right'), ('lower', 'middle')]
        if case == 'offset-stack': node('middle_end', 11, upper_x, 0); edge('middle', 'middle_end')
        node('upper', 12, upper_x, 0); node('end', 12, upper_x, 2.4)
        edge('middle' if case == 'stack' else 'middle_end', 'upper'); edge('upper', 'end')
        near_layer = 11; near_y = 1.6
    near = wire(near_layer, (-1.6, near_y), (4.8, near_y), net='near')
    nodes = dict(target=target, near=[('left', [-.8, near_y, CENTERS[near_layer]]),
                                    ('right', [3.2, near_y, CENTERS[near_layer]])])
    def resistance(net, pair):
        a, b = pair; p, q = (dict(nodes[net])[n] for n in pair)
        if p[2] != q[2]:
            assert p[:2] == q[:2] and abs(abs(p[2]-q[2])-2.8) < 1e-8
            ohm = e['via_resistance_ohm']
        else: ohm = e['sheet_resistance_ohm']*sum(abs(p[k]-q[k]) for k in (0, 1))/g['width']
        return dict(a=a, b=b, ohm=ohm)
    edges = dict(target=[resistance('target', p) for p in links],
                 near=[resistance('near', ('left', 'right'))])
    return dict(boxes=boxes, nodes=nodes, edges=edges, vias=vias,
                routes=dict(target=' NEW '.join(paths), near=near))


def prepare(root, out, cpus, run, case):
    from io_slots import file_sha
    assert cpus == 16
    base = out.parents[2]; config = root/'project/experiments/block_v1/technology.json'
    technology = json.loads(config.read_text()); scene = fixture(case, technology)
    original = base/'block-v1-20261027-009/tasks/route-probe'
    receipt = json.loads((original/'receipt.json').read_text()); assert receipt['passed'] is True
    tech = original/'overlay/research12.tech.lef'
    assert file_sha(tech) == receipt['outputs']['overlay/research12.tech.lef']
    text = tech.read_text()
    for v in scene['vias']:
        name = 'BV1_via%d' % v['layer']
        body = re.search(r'(?ms)^VIA '+name+r' DEFAULT\n(.*?)^END '+name+r'$', text).group(1)
        assert float(re.search(r'RESISTANCE ([\d.]+)', body).group(1)) == technology['electrical']['via_resistance_ohm']
    model = dict(nets=list(scene['nodes']), nodes={n: dict(p) for n, p in scene['nodes'].items()})
    instances = {net+'/'+node: 'E%d' % i for i, (net, node) in enumerate(
        (net, node) for net in model['nets'] for node in model['nodes'][net])}
    native = {n: n.split('/')[0]+'/I:'+inst+'/p' for n, inst in instances.items()}
    anchor_files(out, model, instances); text = (out/'anchors.def').read_text()
    for net, path in scene['routes'].items():
        text, count = re.subn(r'(?m)^- '+net+r' (.*?)\+ USE SIGNAL ;$',
            lambda m: '- '+net+' '+m.group(1)+'+ USE SIGNAL + ROUTED '+path+' ;', text)
        assert count == 1
    (out/'routed-input.def').write_text(text)
    exporter = root/'project/experiments/block_v1/odb_geometry.tcl'
    script = 'set_thread_count 1\nread_lef {'+str(tech)+'}\nread_lef anchors.lef\nread_def routed-input.def\n'
    script += 'write_db routed.odb\nwrite_def routed.def\n'+POSITIONS
    script += 'source {'+str(exporter)+'}\nbvgeom::export geometry.tsv\n'
    (out/'create-routed.tcl').write_text(script); run(['openroad', '-no_init', '-exit', str(out/'create-routed.tcl')], 'create-routed.log')
    actual, kinds = parse((out/'geometry.tsv').read_text(), tech.read_text())
    expected = {net: [b for n, b in scene['boxes'] if n == net] for net in model['nets']}
    union = compare(actual, expected); physical = receive_positions(out/'physical-nodes.tsv', model, native)
    assert physical['passed'] is True and kinds.get('wire-via', 0) > 0
    rejected = []
    for variant in ('old-0.8-cut', 'missing-cuts', 'moved-neighbor'):
        wrong = {net: [b[:] for b in group] for net, group in expected.items()}
        cuts = [b for b in wrong['target'] if any(abs(b[2]-(CENTERS[layer]+1)) < 1e-8
                and abs(b[5]-(CENTERS[layer+1]-1)) < 1e-8 for layer in (10, 11))]
        assert len(cuts) == len(scene['vias'])
        if variant == 'old-0.8-cut':
            for box in cuts:
                for k in (0, 1): box[k] -= .1; box[k+3] += .1
        elif variant == 'missing-cuts': wrong['target'] = [b for b in wrong['target'] if b not in cuts]
        else:
            for box in wrong['near']: box[1] += .1; box[4] += .1
        try: compare(actual, wrong)
        except AssertionError: rejected.append(variant)
        else: raise AssertionError('Accepted corrupt production geometry '+variant)
    # Scientific inputs are exported physical solids, including terminal metal.
    # The separately constructed solid description is only the acceptance oracle.
    boxes = [(net, box) for net in model['nets'] for box in actual[net]]
    inputs = {str(tech): file_sha(tech), str(original/'receipt.json'): file_sha(original/'receipt.json'), str(config): file_sha(config)}
    result = dict(schema='block-v1-production-routed/1', passed=True, case=case, boxes=boxes,
        nodes=scene['nodes'], edges=scene['edges'], vias=scene['vias'], native=native, instances=instances,
        exact_conductor_union=union, physical_nodes=physical, geometry_kinds=kinds, inputs=inputs,
        rejected_geometry_controls=rejected,
        routed_odb_sha256=file_sha(out/'routed.odb'), geometry_sha256=file_sha(out/'geometry.tsv'),
        qualified_scientific_capacitance=False, qualified_manufacturing_drc=False, qualified_full_block=False,
        scope='Actual BV1 0.6um cut/0.1um enclosure geometry and direct physical-node binding only; fresh FV/field and native measured RC acceptance still required')
    (out/'production-routed.json').write_text(json.dumps(result, indent=2)+'\n')


def verify(root, out, cpus, run, case):
    from io_slots import file_sha
    assert cpus == 16 and case in CASES
    base = out.parents[2]; source = root.parent/'tasks'/('production-route-'+case)
    receipt = json.loads((source/'receipt.json').read_text()); assert receipt['passed'] is True
    inputs = {str(source/'receipt.json'): file_sha(source/'receipt.json')}
    for name in ('production-routed.json', 'routed.odb', 'geometry.tsv', 'physical-nodes.tsv'):
        assert file_sha(source/name) == receipt['outputs'][name]; inputs[str(source/name)] = file_sha(source/name)
    for name in ('production_field.py', 'capture_field_verify.py', 'graph_field_verify.py',
                 'joint_rc.py', 'joint_rc_native.py', 'joint_routed_geometry.py', 'joint_node_binding.py', 'odb_geometry.tcl'):
        relative = 'project/experiments/block_v1/'+name
        assert file_sha(root/relative) == receipt['inputs'][relative]
    physical = json.loads((source/'production-routed.json').read_text())
    assert physical['case'] == case and physical['passed'] is True
    assert physical['physical_nodes']['passed'] is True
    assert all(file_sha(Path(p)) == h for p, h in physical['inputs'].items()); inputs.update(physical['inputs'])
    boxes = physical['boxes']; nodes = physical['nodes']
    build = base/'block-v1-20261027-099/tasks/fsv-capture-build'
    bound = json.loads((build/'receipt.json').read_text()); assert bound['passed'] is True
    tool_path = build/'fsv-qualified.json'; assert file_sha(tool_path) == bound['outputs'][tool_path.name]
    tool = json.loads(tool_path.read_text()); binary = Path(tool['capture_binary'])
    assert file_sha(binary) == tool['capture_binary_sha256']
    runtime = base/'block-v1-20261027-033/tasks/fv-install'
    bound = json.loads((runtime/'receipt.json').read_text()); assert bound['passed'] is True
    path = runtime/'fv-runtime.json'; assert file_sha(path) == bound['outputs'][path.name]
    installed = json.loads(path.read_text()); prefix = Path(installed['prefix'])
    for name, digest in installed['files'].items():
        assert file_sha(prefix/name) == digest; inputs[str(prefix/name)] = digest
    for p in (build/'receipt.json', build/'fsv-qualified.json', runtime/'receipt.json', runtime/'fv-runtime.json', binary): inputs[str(p)] = file_sha(p)
    upstream = prefix/'source/fv3d_golden.py'
    (out/'research-joint-fv.py').write_text(fv_source(upstream.read_text()))
    specs = []
    for name, h, far in [('coarse', .4, 80), ('medium', .2, 80), ('fine', .1, 80), ('far', .4, 160)]:
        path = out/(name+'-input.json')
        path.write_text(json.dumps(dict(prefix=str(prefix), source_sha256=file_sha(upstream), boxes=boxes,
            nodes={net: [p for name, p in group] for net, group in nodes.items()}, h=h, far=far,
            output=str(out/(name+'.json'))), indent=2)+'\n'); specs.append(path)
    for origin in nodes:
        (out/(origin+'-scene.txt')).write_text('background 3.9\nmaster '+origin+'\n'+
            ''.join('box '+name+' '+' '.join(map(str, box))+'\n' for name, box in boxes))
        (out/(origin+'-nodes.txt')).write_text(''.join(name+' '+' '.join(map(str, p))+'\n' for name, p in nodes[origin]))
        (out/(origin+'-captures.txt')).write_text(''.join(net+' '+name+' '+' '.join(map(str, p))+'\n'
            for net, group in nodes.items() if net != origin for name, p in group))
    for p in list(specs)+[out/(origin+suffix) for origin in nodes for suffix in ('-scene.txt', '-nodes.txt', '-captures.txt')]: inputs[str(p)] = file_sha(p)
    (out/'production-solver-inputs.json').write_text(json.dumps(dict(inputs=inputs,
        scene_from_actual_odb=True, fresh_geometry=True, thread_budget=16,
        qualified_scientific_capacitance=False), indent=2)+'\n')
    offsets = [.5, .25, .125]; walks = 33554432
    def solve_grid(path):
        run(['env', 'OMP_NUM_THREADS=1', 'OPENBLAS_NUM_THREADS=1', 'MKL_NUM_THREADS=1', 'python3',
             str(root/'project/experiments/block_v1/capture_field_verify.py'), 'grid', str(path)], path.stem+'.log')
    def solve_field(pair):
        origin, offset = pair
        run([str(binary), '2', str(offset), str(walks), str(out/(origin+'-scene.txt')),
             str(out/(origin+'-nodes.txt')), str(out/(origin+'-captures.txt'))], origin+'-offset-'+str(offset)+'.json')
    # Four single-thread FV jobs plus six two-thread walk jobs = exactly 16.
    with concurrent.futures.ThreadPoolExecutor(max_workers=10) as pool:
        futures = [pool.submit(solve_grid, path) for path in specs]
        futures += [pool.submit(solve_field, (origin, offset)) for origin in nodes for offset in offsets]
        for future in futures: future.result()
    sys.path[:0] = [str(prefix/'packages'), str(prefix/'source')]
    grids = [json.loads((out/(name+'.json')).read_text()) for name in ('coarse', 'medium', 'fine', 'far')]
    rows = {origin: [json.loads((out/(origin+'-offset-'+str(offset)+'.json')).read_text()) for offset in offsets] for origin in nodes}
    science = assess(grids, rows, nodes)
    science.update(boxes=boxes, nodes=nodes, grids=grids, field_rows=rows, offsets=offsets, walks_per_offset=walks)
    (out/'production-science.json').write_text(json.dumps(science, indent=2)+'\n')
    assert science['passed'], 'Locked independent production-geometry scientific gates failed'
    model = assemble(nodes, physical['edges'], {n: rows[n][-1] for n in nodes})
    (out/'measured-joint-rc.json').write_text(json.dumps(model, indent=2)+'\n')
    _, instances, native = emit_spef(model)
    assert instances == physical['instances'] and native == physical['native']
    exporter = root/'project/experiments/block_v1/odb_geometry.tcl'
    child = 'import os,sys;os.chdir(sys.argv[1]);os.execvp("openroad",["openroad","-no_init","-exit","receive.tcl"])'
    def ingest(variant):
        folder = out/variant; folder.mkdir()
        spef, current_instances, current_native = emit_spef(model, variant)
        assert current_instances == instances and current_native == native
        (folder/'candidate.spef').write_text(spef)
        script = 'set_thread_count 1\nread_db {'+str(source/'routed.odb')+'}\nbench_read_spef candidate.spef\n'+READBACK+POSITIONS
        script += 'source {'+str(exporter)+'}\nbvgeom::export geometry-after.tsv\nwrite_db received.odb\n'
        (folder/'receive.tcl').write_text(script); run([sys.executable, '-c', child, str(folder)], variant+'.log')
        result = receive(folder, model, native); assert result['passed'] == (variant == 'actual'), (variant, result)
        positions = receive_positions(folder/'physical-nodes.tsv', model, native); assert positions['passed'] is True
        assert file_sha(folder/'geometry-after.tsv') == physical['geometry_sha256']
        return variant, dict(electrical=result, physical=positions, spef_sha256=file_sha(folder/'candidate.spef'),
                             received_odb_sha256=file_sha(folder/'received.odb'))
    with concurrent.futures.ThreadPoolExecutor(max_workers=6) as pool: observed = dict(pool.map(ingest, VARIANTS))
    assert all(file_sha(Path(p)) == h for p, h in inputs.items())
    result = dict(schema='block-v1-production-field-fv/1', passed=True, case=case, inputs=inputs,
        source_physical=physical, independent_science=science, native_readback=observed,
        fv_instrumentation_sha256=file_sha(out/'research-joint-fv.py'),
        thread_budget=dict(fv_jobs=4, fv_threads_each=1, field_jobs=6, field_threads_each=2, total=16),
        resistance_basis='Centerline sheet/width and explicit BV1 via R from sealed research tech; not calibrated manufacturing RC or general junction spreading resistance',
        qualified_two_net_production_geometry=True, qualified_manufacturing_drc=False,
        qualified_general_extractor=False, qualified_boundary_stitch=False, qualified_full_block=False,
        scope='Fresh science and exact native measured RC for this two-net actual 0.6um production-via scene only; no capacitance reuse from sealed0.8um fixture, general graph/window ownership, boundary stitching or full-block claim')
    (out/'production-field-fv.json').write_text(json.dumps(result, indent=2)+'\n')
